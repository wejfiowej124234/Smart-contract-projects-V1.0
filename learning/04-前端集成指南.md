## 1. ethers.js v6 基础

### 1.1 核心概念

```typescript
// Provider：只读，查询区块链状态
const provider = new BrowserProvider(window.ethereum);

// Signer：可以签名交易
const signer = await provider.getSigner();

// Contract：合约实例
const contract = new Contract(address, abi, signer);
```

### 1.2 读操作（免费）

```typescript
// 查询余额
const balance = await contract.balanceOf(userAddress);
// 返回 bigint 类型

// 查询用户仓位
const [supplied, borrowed, collateral, healthFactor] = 
  await contract.getUserPosition(userAddress);
```

### 1.3 写操作（需要 gas）

```typescript
// 发送交易
const tx = await contract.supply(amount);
// 返回 TransactionResponse

// 等待确认
const receipt = await tx.wait(confirmations);
// 返回 TransactionReceipt
```

## 2. 授权（Approve）机制

### 2.1 为什么需要授权？

```typescript
// 你的 USD8 余额：1000
// 借贷合约想转走你的钱，但没权限

// 所以你要先授权：
await usd8Contract.approve(lendingAddress, 100);
// "我允许借贷合约转走我 100 USD8"

// 然后合约才能调用：
await lendingContract.supply(100);
// 合约内部会调用 usd8.transferFrom(你, 合约, 100)
```

### 2.2 检查授权额度

```typescript
const allowance = await usd8Contract.allowance(
  userAddress,      // owner
  lendingAddress    // spender
);

if (allowance < amount) {
  // 需要先 approve
  await usd8Contract.approve(lendingAddress, amount);
}
```

## 3. 事件监听

### 3.1 监听合约事件

```typescript
useEffect(() => {
  if (!lending || !account) return;
  
  const onSupplied = (user: string, amount: bigint, timestamp: bigint) => {
    if (user.toLowerCase() === account.toLowerCase()) {
      console.log(`Supplied ${ethers.formatUnits(amount, 18)} USD8`);
      refresh();  // 刷新 UI
    }
  };
  
  lending.on("Supplied", onSupplied);
  
  // 清理
  return () => {
    lending.off("Supplied", onSupplied);
  };
}, [lending, account]);
```

### 3.2 事件过滤

```typescript
// 只监听特定用户的事件
const filter = lending.filters.Supplied(account);
lending.on(filter, (user, amount, timestamp) => {
  // 只有 user === account 的事件才会触发这里
});
```

### 4.2 合约值 → 显示


### 4.3 计算

```typescript
// ❌ 错误：不要用 Number

const half = amount / 2n;  // 50250000000000000000n
```

## 5. 错误处理

### 5.1 常见错误

```typescript
try {
  await contract.supply(amount);
} catch (err) {
  // 用户拒绝
  if (err.code === "ACTION_REJECTED") {
    console.log("User rejected");
  }
  
  // Gas 不足
  if (err.code === "INSUFFICIENT_FUNDS") {
    console.log("Insufficient gas");
  }
  
  // 合约 revert
  if (err.reason) {
    console.log("Contract reverted:", err.reason);
  }
}
```

### 5.2 错误归一化

```typescript
function normalizeError(err: unknown): NormalizedError {
  if (typeof err === "object" && err !== null) {
    return {
      code: err.code || "UNKNOWN",
      message: err.reason || err.message || "Unknown error",
      method: err.method,
    };
  }
  return {
    code: "UNKNOWN",
    message: String(err),
  };
}
```

## 6. 完整流程示例

### 6.1 Supply 完整流程

```typescript
async function handleSupply(amountStr: string) {
  try {
    // 1. 解析金额
    const amount = ethers.parseUnits(amountStr, 18);
    
    // 2. 检查授权
    const allowance = await usd8.allowance(account, lendingAddress);
    
    if (allowance < amount) {
      // 3. 需要先 approve
      setTx({ stage: "signing", label: "Approving USD8" });
      const approveTx = await usd8.approve(lendingAddress, amount);
      
      setTx({ stage: "pending", label: "Approving USD8", hash: approveTx.hash });
      await approveTx.wait(2);  // 等待 2 次确认
    }
    
    // 4. Supply
    setTx({ stage: "signing", label: "Supplying USD8" });
    const supplyTx = await lending.supply(amount);
    
    setTx({ stage: "pending", label: "Supplying USD8", hash: supplyTx.hash });
    const receipt = await supplyTx.wait(2);
    
    // 5. 验证状态
    const newBalance = await usd8.balanceOf(account);
    
    setTx({ stage: "confirmed", label: "Supplying USD8", hash: receipt.hash });
    
    // 6. 刷新 UI（或者由事件监听自动触发）
    refresh();
    
  } catch (err) {
    setTx({ stage: "failed", label: "Supply", error: normalizeError(err) });
  }
}
```

## 7. 自定义 Hooks 模式

### 7.1 useWallet

```typescript
export function useWallet() {
  const [provider, setProvider] = useState<BrowserProvider>();
  const [account, setAccount] = useState<string>();
  const [chainId, setChainId] = useState<number>();
  
  const connect = async () => {
    const provider = new BrowserProvider(window.ethereum);
    const accounts = await provider.send("eth_requestAccounts", []);
    const network = await provider.getNetwork();
    
    setProvider(provider);
    setAccount(accounts[0]);
    setChainId(Number(network.chainId));
  };
  
  return { provider, account, chainId, connect };
}
```

### 7.2 useDashboard

```typescript
export function useDashboard(provider, account) {
  const [balances, setBalances] = useState({});
  const [position, setPosition] = useState({});
  
  const refresh = useCallback(async () => {
    if (!provider || !account) return;
    
    const contracts = getContracts(provider);
    
    // 并发读取
    const [usd8Balance, supplied, borrowed] = await Promise.all([
      contracts.usd8.balanceOf(account),
      contracts.lending.getUserPosition(account).then(p => p[0]),
      contracts.lending.getUserPosition(account).then(p => p[1]),
    ]);
    
    setBalances({ usd8: usd8Balance });
    setPosition({ supplied, borrowed });
  }, [provider, account]);
  
  // 事件监听
  useEffect(() => {
    const contracts = getContracts(provider);
    contracts.lending.on("Supplied", refresh);
    return () => contracts.lending.off("Supplied", refresh);
  }, [provider, refresh]);
  
  return { balances, position, refresh };
}
```
